# CMake 最低版本要求
cmake_minimum_required(VERSION 3.15)

# ===================================================================
# 项目名称 (这是你为新项目唯一需要修改的地方)
# 这一行会自动创建一个名为 ${PROJECT_NAME} 的变量
project(test)
# ===================================================================

# 设置 C++ 标准为 C++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# --- 添加第三方库 ---

# 添加 GLAD
# GLAD 需要一个头文件目录和一个源文件
include_directories(lib/glad/include)
add_library(glad STATIC lib/glad/src/glad.c)

# 添加 GLFW
# 使用 add_subdirectory 可以让 CMake 构建 GLFW
# 我们不希望构建它的文档和测试，所以关闭这些选项
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(lib/glfw)

# --- 定义我们的可执行文件 ---

# 添加我们的主程序，它由 src/main.cpp 编译而来
file(GLOB_RECURSE SOURCES "src/test04.cpp")

# 将收集到的所有源文件编译成可执行文件
add_executable(${PROJECT_NAME} ${SOURCES})

# --- 链接库 ---

# 为我们的主程序指定需要包含的头文件目录
target_include_directories(${PROJECT_NAME} PUBLIC
    lib/glad/include
    lib/glm
    # GLFW 的头文件目录会通过 target_link_libraries 自动添加
)

# 将我们的主程序链接到它所依赖的库
find_package(OpenGL REQUIRED)

# 链接自动找到的 OpenGL 库 (这句加在你原本的 target_link_libraries 里)
target_link_libraries(${PROJECT_NAME} PUBLIC
    glad
    glfw
    OpenGL::GL  # <--- 加这一行，这是完全跨平台、跨编译器的写法
)
